POST /register
    receive userDTO
    hash password
    save to UserRepository

POST /login
    receive email & password
    find user by email
    if password matches:
        generate JWT token
        return user info + token
    else return 401

GET /logout
    invalidate token or delete session
    return success

GET /profile
    validate token
    return current user data

POST /profile/update/profile
    validate token
    receive namabaru, email, alamat
    if namabaru != null && email != null && email == valid && alamat != null
        update profile
    else
        return 401
    

POST /profile/update/password
    validate token
    receive passwordlama, passwordbaru, konfirmasipasswordbaru
    if passwordbaru == konfirmasipasswordbaru
        update password
    else
        Password gagal diubah
